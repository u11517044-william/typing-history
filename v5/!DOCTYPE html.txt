    // 擊退：每字微幅；完成單字再大幅擊退（單字越長退越遠）
    var kb = KB_SMALL;
    var bigKb = 0;
    if (unitDone) bigKb = len > 1 ? Math.min(4 + len * 0.9, 24) : 1.0;
    kb += bigKb;
    G.bossPos = Math.max(0, G.bossPos - kb);

    var before = G.bossHp;
    G.bossHp -= dmg;
    G.totalDmg += Math.min(dmg, before);
    var dead = G.bossHp <= 0;

    fireMissile(unitDone ? 'big' : '');
    setTimeout(function () {
      retrigger($('boss'), 'hit');
      floatText('bossIcon', (unitDone ? '爆擊 -' : '-') + dmg, unitDone ? 'crit' : '');
      if (bigKb >= 4) { ringAt('bossIcon'); floatText('bossIcon', '◀ 擊退！', 'kb'); }
      updateBossBar();
    }, 210);

    if (unitDone) setLog('💥 完整擊破「' + G.text + '」！爆擊 ' + dmg + '，Boss 被大幅擊退！（×' + mult.toFixed(2) + '）');

    if (dead) { G.state = 'over'; updateTarget(); renderHud(); win(); return; }

    if (unitDone) {
      retrigger($('target'), 'flash');
      G.text = nextUnit(); G.pos = 0;
      renderUnit();
    } else {
      updateTarget();
    }
  }

    if (G.state === 'playing') {
      G.elapsed += dt;
      G.bossPos = Math.min(100, G.bossPos + F.speed * dt);      // Boss 由左向右推進
      var contact = G.bossPos >= 100;
      if (contact) {
        G.contactTimer += dt;
        if (G.contactTimer >= 1) { G.contactTimer -= 1; contactTick(); }
      } else {
        G.contactTimer = 0;
      }
      if (G.state === 'playing') {
        $('warn').classList.toggle('on', contact);
        $('warnText').classList.toggle('on', contact);
        $('crystal').classList.toggle('danger', contact);
      }
    }
    G.bossVis += (G.bossPos - G.bossVis) * Math.min(1, dt * 10); // 擊退平滑動畫
    layoutBoss();
    renderHud();
    requestAnimationFrame(loop);
  }

  /* ================= 勝敗 ================= */
  function endWarn() {
    $('warn').classList.remove('on');
    $('warnText').classList.remove('on');
    $('crystal').classList.remove('danger');
  }
  function win() {
    var F = FLOORS[G.floor - 1];
    endWarn();
    if (progress.cleared.indexOf(G.floor) === -1) progress.cleared.push(G.floor);
    if (G.floor < FLOORS.length && progress.unlocked < G.floor + 1) progress.unlocked = G.floor + 1;
    setLog('🏆 擊敗了 ' + F.name + '！');
    setTimeout(function () { $('boss').classList.add('dead'); }, 230);
    setTimeout(function () {
      saveProgress();
      renderTower();
      showModal(true);
    }, 1200);
  }
  function lose() {
    G.state = 'over';
    var F = FLOORS[G.floor - 1];
    endWarn();
    $('player').classList.add('dead');
    $('crystal').classList.add('dead');
    setLog('💀 水晶被 ' + F.name + ' 摧毀了…');
    updateKeyboard();
    setTimeout(function () { showModal(false); }, 800);
  }

  /* ================= 彈窗 ================= */
  var modalOpen = false;
  function showModal(victory) {
    var F = FLOORS[G.floor - 1];
    var last = G.floor >= FLOORS.length;
    var total = G.correct + G.wrong;
    var acc = total ? (G.correct / total * 100).toFixed(1) + '%' : '--';
    var dps = G.elapsed > 0.5 ? Math.round(G.totalDmg / G.elapsed) : 0;
    var m = Math.floor(G.elapsed / 60), s = Math.floor(G.elapsed % 60);
    var time = (m < 10 ? '0' : '') + m + ':' + (s < 10 ? '0' : '') + s;
    var box = $('modalBox');
    box.className = 'box' + (victory ? '' : ' fail');
    var title = victory ? (last ? '👑 登頂成功！' : '💥 擊殺通關！') : '💀 挑戰失敗';
    var msg = victory
      ? (last ? '你擊敗了高塔終極魔王，征服了整座高塔！' : '擊敗了 ' + F.name + '，已解鎖第 ' + (G.floor + 1) + ' 層！')
      : '水晶被摧毀了，穩住節奏、少打錯字再挑戰一次吧！';
    box.innerHTML =
      '<h2>' + title + '</h2><p>' + msg + '</p>' +
      '<div class="res">' +
        '<div><b>' + time + '</b><span>戰鬥時間</span></div>' +
        '<div><b>' + dps + '</b><span>DPS</span></div>' +
        '<div><b>' + acc + '</b><span>正確率</span></div>' +
        '<div><b>' + G.maxCombo + '</b><span>最高連擊</span></div>' +
      '</div>' +
      '<div class="btns">' +
        (victory && !last ? '<button class="btn primary" id="mNext" type="button">前往下一層 ▶</button>' : '') +
        '<button class="btn' + (victory && !last ? '' : ' primary') + '" id="mRetry" type="button">' + (victory ? '重新挑戰' : '再戰一次') + '</button>' +
        '<button class="btn" id="mClose" type="button">關閉</button>' +
      '</div>';
    $('modal').hidden = false;
    modalOpen = true;
    var nx = $('mNext');
    if (nx) nx.addEventListener('click', function () { loadFloor(G.floor + 1); });
    $('mRetry').addEventListener('click', function () { loadFloor(G.floor); });
    $('mClose').addEventListener('click', closeModal);
  }
  function closeModal() {
    $('modal').hidden = true;
    modalOpen = false;
    if (document.activeElement && document.activeElement.blur) document.activeElement.blur();
  }

  /* ================= 重置 ================= */
  $('resetBtn').addEventListener('click', function () {
    if (confirm('確定要重置所有遊戲進度嗎？此動作無法復原。')) {
      progress = { unlocked: 1, cleared: [] };
      saveProgress();
      loadFloor(1);
    }
  });

  /* ================= 啟動 ================= */
  buildKeyboard();
  loadFloor(progress.unlocked);
  requestAnimationFrame(loop);
})();
